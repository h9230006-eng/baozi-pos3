function updateSummary() { const daysInMonth = getDaysInCurrentMonth(); const monthKey = getMonthKey(); let totalMonthlyRest = 0; staffList.forEach(staff => { let restDays = 0; for (let d = 1; d <= daysInMonth; d++) { const val = scheduleData[monthKey]?.[staff.name]?.[d] || ""; if (val === '休') { restDays++; } } totalMonthlyRest += restDays; const restCountEl = document.getElementById(`rest_count_${staff.name}`); if (restCountEl) { restCountEl.innerText = `${restDays}天`; } }); for (let d = 1; d <= daysInMonth; d++) { let restCount = 0; staffList.forEach(staff => { const val = scheduleData[monthKey]?.[staff.name]?.[d] || ""; if (val === '休') { restCount++; } }); // 原本當日上班人數 let workCount = staffList.length - restCount; // 經理點選「支援」時，當日上班人數額外增加 1 人 const managerSupport = scheduleData[monthKey]?.['經理']?.[d] === '支援'; if (managerSupport) { workCount += 1; } const restCell = document.getElementById(`summary_rest_${d}`); const workCell = document.getElementById(`summary_work_${d}`); if (restCell) { restCell.innerText = `${restCount}人`; } if (workCell) { workCell.innerText = `${workCount}人`; } } const totalRestCell = document.getElementById('summary_total_rest'); if (totalRestCell) { totalRestCell.innerText = `${totalMonthlyRest}天`; } }
